# ============================================================================
# Inner First Aid — OPTIONAL server optimization for Apache / LiteSpeed hosts
# ----------------------------------------------------------------------------
# HOW TO USE:
#   1. Open your host's File Manager (cPanel/hPanel).
#   2. Find the .htaccess file next to wp-config.php (site root).
#   3. Open it in the editor. You will see a section starting with
#      "# BEGIN WordPress". DO NOT touch that section.
#   4. Paste the blocks below AFTER the "# END WordPress" line and save.
#
# NOTE: Security headers (X-Frame-Options etc.) are already sent by the theme
# itself, so they are NOT included here to avoid duplicates.
# ============================================================================

# --- 1. Browser caching (make images/CSS/JS load instantly on repeat visits) ---
<IfModule mod_expires.c>
	ExpiresActive On
	ExpiresByType image/jpeg "access plus 1 year"
	ExpiresByType image/png "access plus 1 year"
	ExpiresByType image/gif "access plus 1 year"
	ExpiresByType image/webp "access plus 1 year"
	ExpiresByType image/svg+xml "access plus 1 year"
	ExpiresByType image/x-icon "access plus 1 year"
	ExpiresByType font/woff "access plus 1 year"
	ExpiresByType font/woff2 "access plus 1 year"
	ExpiresByType text/css "access plus 1 month"
	ExpiresByType application/javascript "access plus 1 month"
	ExpiresByType text/javascript "access plus 1 month"
	ExpiresByType application/json "access plus 0 seconds"
	ExpiresByType text/html "access plus 0 seconds"
</IfModule>

<IfModule mod_headers.c>
	<FilesMatch "\.(jpg|jpeg|png|gif|webp|svg|ico|css|js|woff|woff2)$">
		Header set Cache-Control "public, max-age=31536000, immutable"
	</FilesMatch>
</IfModule>

# --- 2. Compression (smaller pages = faster loads) ---
<IfModule mod_deflate.c>
	AddOutputFilterByType DEFLATE text/html text/plain text/css text/xml
	AddOutputFilterByType DEFLATE application/javascript application/json
	AddOutputFilterByType DEFLATE application/xml image/svg+xml
</IfModule>

# --- 3. Protect wp-config.php from direct access ---
<Files wp-config.php>
	<IfModule mod_authz_core.c>
		Require all denied
	</IfModule>
	<IfModule !mod_authz_core.c>
		Order allow,deny
		Deny from all
	</IfModule>
</Files>

# --- 4. Disable directory listing ---
Options -Indexes
